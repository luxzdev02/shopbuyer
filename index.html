OTP Otomatis Terdeteksi

Sistem kami auto-cek tiap 10 detik dan kirim OTP ke kamu langsung.

quick-start.js 
// Setup axios instance
const api = axios.create({
  baseURL: 'https://fnbnokos.web.id',
  headers: { 'x-api-key': 'YOUR_API_KEY' }
});

// 1. Order nomor OTP
const order = await api.post('/api/v1/s2/nokos/order', {
  service_code: 'wa',
  country_id:   62,
  server:       's2'
});

// 2. Ambil nomor dari response
const { id, phone_number } = order.data.order;
console.log(`Nomor: ${phone_number}`);

// 3. Polling status OTP
const res = await api.get('/api/v1/s2/nokos/status', {
  params: { order_id: id }
});
console.log(`OTP: ${res.data.otp_code}`);
// → OTP: 847291 ✓
